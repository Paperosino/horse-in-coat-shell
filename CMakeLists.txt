cmake_minimum_required(VERSION 3.27)
project(hics VERSION 1.1.1 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_INSTALL_LIBDIR "${CMAKE_INSTALL_DATADIR}/hics/lib" CACHE PATH "Private runtime libraries" FORCE)

set(VERSION "${PROJECT_VERSION}")
set(GIT_REVISION "release")
set(DISTRIBUTOR "hics")
set(INSTALL_LIBDIR "${CMAKE_INSTALL_DATADIR}/hics/lib" CACHE STRING "Helper directory" FORCE)
set(INSTALL_QMLDIR "${CMAKE_INSTALL_DATADIR}/hics/qml" CACHE STRING "Private QML modules" FORCE)
set(INSTALL_QSCONFDIR "${CMAKE_INSTALL_DATADIR}/hics/shell" CACHE STRING "Shell directory" FORCE)
option(HYPRSTACK_BUILD_CANVAS "Build the pinned Phantomat canvas plugin" ON)
option(HYPRSTACK_BUILD_NATIVE "Build matching compositor extensions" ON)
option(HYPRSTACK_BUNDLE_RUNTIME "Build a private quickshell+Qt runtime" OFF)
set(HYPRSTACK_QUICKSHELL_TAG "v0.3.2" CACHE STRING "Pinned quickshell version")

# Public hics build options; retain compatibility with existing build caches.
foreach(option BUILD_NATIVE BUILD_CANVAS BUNDLE_RUNTIME AUDIO_VISUALIZER)
    if(DEFINED HICS_${option})
        set(HYPRSTACK_${option} "${HICS_${option}}" CACHE BOOL "hics build option" FORCE)
    endif()
endforeach()
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Qml Quick ShaderTools)
set(M3SHAPES_INSTALL_DEVELOPMENT OFF CACHE BOOL "Install development files" FORCE)
add_subdirectory(third_party/m3shapes)
add_subdirectory(shell)
add_subdirectory(native/system)

if(HYPRSTACK_BUILD_NATIVE)
    execute_process(COMMAND Hyprland --version-json OUTPUT_VARIABLE hypr_version
        RESULT_VARIABLE version_result OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT version_result EQUAL 0)
        message(FATAL_ERROR "Install Hyprland 0.56.2 and matching development headers first")
    endif()
    string(JSON hypr_tag ERROR_VARIABLE version_error GET "${hypr_version}" version)
    if(version_error OR NOT hypr_tag STREQUAL "0.56.2")
        message(FATAL_ERROR "This native extension supports Hyprland 0.56.2. See INSTALL.md")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(Native REQUIRED IMPORTED_TARGET hyprland pixman-1 libdrm libinput libudev wayland-server xkbcommon pangocairo)
    find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
    string(JSON hypr_abi ERROR_VARIABLE abi_error GET "${hypr_version}" abiHash)
    if(abi_error)
        message(FATAL_ERROR "Hyprland must report its native plugin ABI")
    endif()
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/distribution/check_headers.py" "${hypr_abi}"
        RESULT_VARIABLE headers_result OUTPUT_VARIABLE headers_output ERROR_VARIABLE headers_error)
    if(NOT headers_result EQUAL 0)
        message(FATAL_ERROR "${headers_error}")
    endif()
    message(STATUS "${headers_output}")
    add_library(hyprstack-bars MODULE native/hyprbars/main.cpp native/hyprbars/barDeco.cpp native/hyprbars/BarPassElement.cpp)
    set_target_properties(hyprstack-bars PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON PREFIX "" OUTPUT_NAME "hics-bars")
    target_link_libraries(hyprstack-bars PRIVATE PkgConfig::Native)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(hyprstack-bars PRIVATE -fno-gnu-unique)
    endif()
    target_compile_options(hyprstack-bars PRIVATE -Wno-narrowing)
    install(TARGETS hyprstack-bars LIBRARY DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/native")
    add_library(hics-capture MODULE native/capture/main.cpp)
    set_target_properties(hics-capture PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON PREFIX "")
    target_link_libraries(hics-capture PRIVATE PkgConfig::Native)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(hics-capture PRIVATE -fno-gnu-unique)
    endif()
    install(TARGETS hics-capture LIBRARY DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/native")
    add_library(hics-motion MODULE native/motion/main.cpp)
    set_target_properties(hics-motion PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON PREFIX "")
    target_link_libraries(hics-motion PRIVATE PkgConfig::Native)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(hics-motion PRIVATE -fno-gnu-unique)
    endif()
    install(TARGETS hics-motion LIBRARY DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/native")
    if(HYPRSTACK_BUILD_CANVAS)
        add_subdirectory(third_party/phantomat)
        install(TARGETS spatialoverview LIBRARY DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/native")
        install(FILES third_party/phantomat/LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/licenses" RENAME Phantomat-BSD-3-Clause)
    endif()
    file(WRITE "${CMAKE_BINARY_DIR}/native-version.json" "${hypr_version}\n")
    install(FILES "${CMAKE_BINARY_DIR}/native-version.json" DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/native")
endif()

if(HYPRSTACK_BUNDLE_RUNTIME)
    include(cmake/BundleRuntime.cmake)
endif()

install(DIRECTORY distribution/defaults/ DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/defaults")
install(PROGRAMS distribution/bin/hics distribution/bin/hics-shell distribution/bin/hics-session distribution/bin/hics-session-runner distribution/bin/hics-doctor DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES distribution/systemd/user/hic-session.service DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/systemd")
install(FILES shell/LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/hics" RENAME LICENSE)
install(FILES THIRD_PARTY.md DESTINATION "${CMAKE_INSTALL_DATADIR}/hics")
install(FILES native/hyprbars/LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/licenses" RENAME hyprbars-BSD-3-Clause)
install(FILES third_party/m3shapes/LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/licenses" RENAME M3Shapes-Apache-2.0)
install(FILES shell/assets/logo.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps" RENAME hyprstack.svg)

install(DIRECTORY distribution/licenses/ DESTINATION "${CMAKE_INSTALL_DATADIR}/hics/licenses")
