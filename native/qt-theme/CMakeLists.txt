# This platform adapter belongs to host applications, not the bundled shell Qt.
find_package(Qt6 REQUIRED COMPONENTS Gui)
find_package(Qt6GuiPrivate REQUIRED)
add_library(hics-qt6-theme MODULE main.cpp)
target_link_libraries(hics-qt6-theme PRIVATE Qt6::Gui Qt6::GuiPrivate)
set_target_properties(hics-qt6-theme PROPERTIES AUTOMOC ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt-themes/qt6/platformthemes")
install(TARGETS hics-qt6-theme LIBRARY DESTINATION "lib/qt6/plugins/platformthemes")
find_package(Qt5 5.15 QUIET COMPONENTS Gui)
if(TARGET Qt5::GuiPrivate)
    add_library(hics-qt5-theme MODULE main.cpp)
    target_link_libraries(hics-qt5-theme PRIVATE Qt5::Gui Qt5::GuiPrivate)
    set_target_properties(hics-qt5-theme PROPERTIES AUTOMOC ON LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qt-themes/qt5/platformthemes")
    install(TARGETS hics-qt5-theme LIBRARY DESTINATION "lib/qt/plugins/platformthemes")
endif()
