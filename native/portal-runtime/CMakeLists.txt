cmake_minimum_required(VERSION 3.27)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(HicsPortalRuntime LANGUAGES CXX)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(Systemd REQUIRED IMPORTED_TARGET libsystemd)
add_library(hics-portal-loop STATIC eventloop.cpp process.cpp)
target_compile_features(hics-portal-loop PUBLIC cxx_std_23)
target_link_libraries(hics-portal-loop PUBLIC PkgConfig::Systemd)
target_include_directories(hics-portal-loop PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(hics-portal-loop PRIVATE -Wall -Wextra -Werror)
