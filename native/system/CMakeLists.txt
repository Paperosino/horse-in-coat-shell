find_package(Qt6 REQUIRED COMPONENTS Core DBus)
find_package(Cups REQUIRED)
add_library(hyprstack-system-backend STATIC backend.cpp backend.hpp storage.cpp)
set_target_properties(hyprstack-system-backend PROPERTIES CXX_STANDARD 20)
target_link_libraries(hyprstack-system-backend PUBLIC Qt6::Core Qt6::DBus Cups::Cups)
# Resolve fixed executables at build time, including non-/usr Nix installations.
foreach(tool systemd-run systemctl fstrim e4defrag)
  string(TOUPPER "${tool}" tool_key)
  string(REPLACE "-" "_" tool_key "${tool_key}")
  find_program(HYPRSTACK_${tool_key} NAMES ${tool})
  if(NOT HYPRSTACK_${tool_key})
    set(HYPRSTACK_${tool_key} "")
  endif()
  target_compile_definitions(hyprstack-system-backend PRIVATE HYPRSTACK_${tool_key}="${HYPRSTACK_${tool_key}}")
endforeach()
add_executable(hyprstack-systemd main.cpp)
set_target_properties(hyprstack-systemd PROPERTIES AUTOMOC ON CXX_STANDARD 20)
target_link_libraries(hyprstack-systemd PRIVATE hyprstack-system-backend)
add_executable(hyprstack-systemctl client.cpp)
set_target_properties(hyprstack-systemctl PROPERTIES CXX_STANDARD 20)
target_link_libraries(hyprstack-systemctl PRIVATE hyprstack-system-backend)
set_target_properties(hyprstack-systemd hyprstack-systemctl PROPERTIES INSTALL_RPATH "$ORIGIN/../share/hyprstack/qs/lib" INSTALL_RPATH_USE_LINK_PATH ON)
install(TARGETS hyprstack-systemd hyprstack-systemctl RUNTIME DESTINATION bin)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../distribution/systemd/system/hyprstack-system.service" system_unit)
string(REPLACE "/usr/bin/hyprstack-systemd" "${CMAKE_INSTALL_PREFIX}/bin/hyprstack-systemd" system_unit "${system_unit}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hyprstack-system.service" "${system_unit}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hyprstack-system.service" DESTINATION lib/systemd/system)
install(FILES ../../distribution/dbus/org.hyprstack.System1.conf DESTINATION share/dbus-1/system.d)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../distribution/dbus/org.hyprstack.System1.service" dbus_unit)
string(REPLACE "/usr/bin/hyprstack-systemd" "${CMAKE_INSTALL_PREFIX}/bin/hyprstack-systemd" dbus_unit "${dbus_unit}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/org.hyprstack.System1.service" "${dbus_unit}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.hyprstack.System1.service" DESTINATION share/dbus-1/system-services)
install(FILES ../../distribution/polkit/org.hyprstack.system.policy DESTINATION share/polkit-1/actions)
