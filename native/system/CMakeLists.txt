find_package(Qt6 REQUIRED COMPONENTS Core DBus)
find_package(Cups REQUIRED)
add_library(hics-system-backend STATIC backend.cpp backend.hpp storage.cpp)
set_target_properties(hics-system-backend PROPERTIES CXX_STANDARD 20)
target_link_libraries(hics-system-backend PUBLIC Qt6::Core Qt6::DBus Cups::Cups)
# Resolve fixed executables at build time, including non-/usr Nix installations.
foreach(tool systemd-run systemctl fstrim e4defrag)
  string(TOUPPER "${tool}" tool_key)
  string(REPLACE "-" "_" tool_key "${tool_key}")
  find_program(HYPRSTACK_${tool_key} NAMES ${tool})
  if(NOT HYPRSTACK_${tool_key})
    set(HYPRSTACK_${tool_key} "")
  endif()
  target_compile_definitions(hics-system-backend PRIVATE HYPRSTACK_${tool_key}="${HYPRSTACK_${tool_key}}")
endforeach()
add_executable(hics-systemd main.cpp)
set_target_properties(hics-systemd PROPERTIES AUTOMOC ON CXX_STANDARD 20)
target_link_libraries(hics-systemd PRIVATE hics-system-backend)
add_executable(hics-systemctl client.cpp)
set_target_properties(hics-systemctl PROPERTIES CXX_STANDARD 20)
target_link_libraries(hics-systemctl PRIVATE hics-system-backend)
set_target_properties(hics-systemd hics-systemctl PROPERTIES INSTALL_RPATH "$ORIGIN/../share/hics/qs/lib" INSTALL_RPATH_USE_LINK_PATH ON)
install(TARGETS hics-systemd hics-systemctl RUNTIME DESTINATION bin)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../distribution/systemd/system/hics-system.service" system_unit)
string(REPLACE "/usr/bin/hics-systemd" "${CMAKE_INSTALL_PREFIX}/bin/hics-systemd" system_unit "${system_unit}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hics-system.service" "${system_unit}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hics-system.service" DESTINATION lib/systemd/system)
install(FILES ../../distribution/dbus/org.hics.System1.conf DESTINATION share/dbus-1/system.d)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../distribution/dbus/org.hics.System1.service" dbus_unit)
string(REPLACE "/usr/bin/hics-systemd" "${CMAKE_INSTALL_PREFIX}/bin/hics-systemd" dbus_unit "${dbus_unit}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/org.hics.System1.service" "${dbus_unit}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.hics.System1.service" DESTINATION share/dbus-1/system-services)
install(FILES ../../distribution/polkit/org.hics.system.policy DESTINATION share/polkit-1/actions)
